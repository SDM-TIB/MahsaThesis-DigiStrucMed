% --- packages (most are probably already loaded by your template; loading
%     them twice is harmless) ---------------------------------------------

% --- Definition box (dark header, like "Definition 1" in your PDF) ----
\newtcolorbox[auto counter]{bgdefinition}[1]{%
  enhanced, breakable,
  colback=white, colframe=black!85,
  coltitle=white, fonttitle=\bfseries,
  arc=1.2mm, boxrule=0.6pt,
  left=2mm, right=2mm, top=1.5mm, bottom=1.5mm,
  title={Definition~\thetcbcounter:~#1}}

% --- Example box (thin grey frame, bold "Example N" first line) ------
\newtcolorbox[auto counter]{bgexample}[1][]{%
  enhanced, breakable,
  colback=white, colframe=black!45,
  sharp corners, boxrule=0.5pt,
  left=3mm, right=3mm, top=2mm, bottom=2mm,
  before upper={\textbf{Example~\thetcbcounter}%
                \if\relax\detokenize{#1}\relax\else\textbf{:~#1}\fi\par\smallskip}}

% --- Continue numbering after the boxes you already have -------------
% Your Background chapter already contains "Definition 1" and "Example 1",
% so the new boxes start at 2. If you add more of your own boxes before
% these files, raise the numbers below (or delete the three lines).
\makeatletter
\setcounter{tcb@cnt@bgdefinition}{1}\setcounter{tcb@cnt@bgexample}{1}
\makeatother
